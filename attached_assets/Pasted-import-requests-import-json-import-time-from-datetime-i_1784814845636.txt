# 3. Motores de Agregação Pública e Deep Research (Sem uso de Tokens)
def buscar_cisa_kev(tecnologia):
    url = "https://www.cisa.gov/sites/default/files/feeds/known_exploited_vulnerabilities.json"
    resultados = []
    try:
        resp = requests.get(url, timeout=10)
        if resp.status_code == 200:
            dados = resp.json()
            for vuln in dados.get("vulnerabilities", []):
                data_add = datetime.strptime(vuln.get("dateAdded"), "%Y-%m-%d")
                if (datetime.now() - data_add).days <= 30:
                    if tecnologia.lower() in vuln.get("vendorProject", "").lower() or tecnologia.lower() in vuln.get("product", "").lower():
                        resultados.append({
                            "id": vuln.get("cveID"),
                            "tech": tecnologia,
                            "desc": vuln.get("shortDescription"),
                            "solucao": vuln.get("requiredAction"),
                            "cvss": "N/D (Exploração Ativa)",
                            "fonte": "CISA KEV"
                        })
    except Exception:
        pass
    return resultados

def buscar_osv_dev(tecnologia):
    url = "https://api.osv.dev/v1/query"
    payload = {"package": {"name": tecnologia.lower()}}
    resultados = []
    try:
        resp = requests.post(url, json=payload, timeout=10)
        if resp.status_code == 200:
            dados = resp.json()
            for vuln in dados.get("vulns", [])[:2]:
                resultados.append({
                    "id": vuln.get("id", "OSV-VULN"),
                    "tech": tecnologia,
                    "desc": vuln.get("details", vuln.get("summary", "Detalhes técnicos fornecidos na base OSV.")),
                    "solucao": "Atualizar biblioteca/pacote afetado no repositório.",
                    "cvss": "N/D",
                    "fonte": "OSV.dev (Open Source)"
                })
    except Exception:
        pass
    return resultados

def buscar_circl(tecnologia):
    # Consulta a base europeia CIRCL
    url = f"https://cve.circl.lu/api/search/{tecnologia.lower()}"
    resultados = []
    try:
        resp = requests.get(url, timeout=10)
        if resp.status_code == 200:
            dados = resp.json()
            for vuln in dados.get("data", [])[:2]:
                resultados.append({
                    "id": vuln.get("id"),
                    "tech": tecnologia,
                    "desc": vuln.get("summary", "Sem descrição disponível."),
                    "solucao": "Verificar boletins do fabricante.",
                    "cvss": str(vuln.get("cvss", "N/D")),
                    "fonte": "CIRCL CVE Search"
                })
    except Exception:
        pass
    return resultados

def buscar_nvd(tecnologia, dias=2):
    hoje = datetime.now(timezone.utc)
    inicio = hoje - timedelta(days=dias)
    url = "https://services.nvd.nist.gov/rest/json/cves/2.0"
    params = {
        "pubStartDate": inicio.strftime("%Y-%m-%dT%H:%M:%S.000Z"),
        "pubEndDate": hoje.strftime("%Y-%m-%dT%H:%M:%S.000Z"),
        "keywordSearch": tecnologia,
        "resultsPerPage": 2
    }
    resultados = []
    try:
        time.sleep(6) # Pausa obrigatória para IP sem token no NVD (Evita bloqueio 403)
        resp = requests.get(url, params=params, timeout=15)
        if resp.status_code == 200:
            dados = resp.json()
            for item in dados.get("vulnerabilities", []):
                cve = item.get("cve", {})
                cvss = "N/D"
                metrics = cve.get("metrics", {})
                if "cvssMetricV31" in metrics:
                    cvss = str(metrics["cvssMetricV31"].get("cvssData", {}).get("baseScore", "N/D"))

                desc = cve.get("descriptions", [{}])[0].get("value", "Sem descrição disponível.")
                resultados.append({
                    "id": cve.get("id"),
                    "tech": tecnologia,
                    "desc": desc,
                    "solucao": "Aplicar atualizações de segurança fornecidas pelo fabricante ou rotacionar credenciais afetadas.",
                    "cvss": cvss,
                    "fonte": "NVD / NIST"
                })
    except Exception:
        pass
    return resultados

out = widgets.Output()
lbl_titulo = widgets.HTML("<h2>Deep Research de Vulnerabilidades (Multi-bases)</h2>")

# Cria uma interface limpa onde você pode selecionar todos (CTRL+A) ou específicos
seletor_ambientes = widgets.SelectMultiple(
    options=AMBIENTES,
    value=list(AMBIENTES), # Já vem com todos selecionados por padrão
    description='Sistemas:',
    disabled=False,
    layout=widgets.Layout(width='50%', height='300px')
)

btn_buscar = widgets.Button(description="Iniciar Varredura Profunda", button_style='primary')

def gerar_sugestao(cvss):
    try:
        if float(cvss) >= 9.0: return "Ação Emergencial Recomendada"
        if float(cvss) >= 7.0: return "Aplicar Correção na próxima janela"
    except:
        pass
    return "Monitorar e Planejar Atualização"

def on_btn_buscar_clicked(b):
    with out:
        clear_output()
        selecionados = seletor_ambientes.value
        print(f"Iniciando varredura em NVD, CISA KEV, CIRCL e OSV.dev para {len(selecionados)} tecnologias.")
        print("Aguarde... As pausas entre as requisições (rate-limit) previnem o bloqueio do seu Colab.\n")

        cves_encontradas = []
        for tech in selecionados:
            resultados = buscar_cisa_kev(tech) + buscar_osv_dev(tech) + buscar_circl(tech) + buscar_nvd(tech)
            for cve in resultados:
                if cve['id'] not in db["processadas"]:
                    if not any(d['id'] == cve['id'] for d in cves_encontradas):
                        cves_encontradas.append(cve)

        clear_output()
        if not cves_encontradas:
            print("✅ Nenhuma vulnerabilidade nova não tratada para os sistemas selecionados hoje.")
            return

        print(f"⚠️ {len(cves_encontradas)} vulnerabilidades encontradas aguardando triagem.\n")

        for cve in cves_encontradas:
            display(HTML(f"<h3>{cve['id']} ({cve['tech']})</h3>"))
            display(HTML(f"<b>CVSS:</b> {cve['cvss']} | <b>Fonte:</b> {cve['fonte']} | <i>{gerar_sugestao(cve['cvss'])}</i>"))
            display(HTML(f"<b>Descrição:</b> {cve['desc']} ")) # Added description display during triage

            acao = widgets.RadioButtons(
                options=['Processar', 'Adiar', 'Descartar'],
                description='Decisão:',
                layout=widgets.Layout(width='100%')
            )
            btn_aplicar = widgets.Button(description="Confirmar", button_style='info')

            def criar_callback(cve_dict, acao_widget):
                def aplicar(b):
                    escolha = acao_widget.value
                    cve_id = cve_dict['id']

                    if "Processar" in escolha or "Descartar" in escolha:
                        if cve_id not in db["processadas"]:
                            db["processadas"].append(cve_id)
                            if cve_id in db["adiadas"]:
                                db["adiadas"].remove(cve_id)
                    elif "Adiar" in escolha:
                        if cve_id not in db["adiadas"]:
                            db["adiadas"].append(cve_id)

                    save_db(db)
                    b.description = "Registrado!"
                    b.button_style = 'success'
                    b.disabled = True

                    if "Processar" in escolha:
                        display(HTML(gerar_html_tenable(cve_dict)))

                return aplicar

            btn_aplicar.on_click(criar_callback(cve, acao))
            display(acao, btn_aplicar)
            display(HTML("<hr>"))

btn_buscar.on_click(on_btn_buscar_clicked)
display(lbl_titulo, seletor_ambientes, btn_buscar, out)
